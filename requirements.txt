# BONSAI - recommended Python packages.
# EVERY line below is optional: the app runs on the standard library alone and
# each tool reports a clear error if its package is missing. This file installs
# them all so every tool family is live (see requirements.md for the mapping).
#
# Screen capture / input / clipboard  -> take_screenshot, control_input, clipboard
pyautogui
pillow
pyperclip

# Window management                 -> window_list, window_action (Windows only)
pywin32; sys_platform == "win32"
psutil

# API client + WebSocket            -> api_call, ws_test
requests
websocket-client

# Helper scripts                     -> tools/scraper.py, tools/ocr.py
beautifulsoup4
pytesseract

# Blender control (optional addon)   -> Blender MCP tools
mcp-for-blender

# Headless browser                   -> web_resolve(render:true)
# Installed but never declared here, so `pip install -r requirements.txt` gave
# you no browser and the resolver quietly answered "no browser available". Needs
# a one-off `playwright install chromium` as well.
playwright

# Local neural TTS                   -> tts_speak, tts_voices, /kokoro (Kokoro-82M)
# kokoro-onnx pulls espeakng-loader and phonemizer, which it needs to turn text
# into the phonemes the model was trained on. The weights are separate and are
# NOT here: run `python kokoro\download_model.py` once (~340 MB).
kokoro-onnx
onnxruntime
numpy
sounddevice
